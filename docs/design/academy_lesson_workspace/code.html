Building a React Component

schedule 20 min bar_chart Intermediate code React

In this lesson, we will construct a high-performance, generative React component that integrates with the CASE architecture. You will learn how to structure internal state, map UI nodes to generative models, and handle complex asynchronous rendering cycles.

Component Architecture

A generative node component must strictly separate layout logic from data hydration. We leverage React.Suspense alongside tailored context providers to achieve this.

GenerativeNode.tsx 
import React, { useState, useEffect } from 'react';
import { useGenerativeModel } from '@case/core';

interface NodeProps {
  nodeId: string;
  schemaVersion: number;
}

export const GenerativeNode: React.FC<NodeProps> = ({ nodeId, schemaVersion }) => {
  const { hydrate, isProcessing } = useGenerativeModel(nodeId);
  const [canvasState, setCanvasState] = useState<any>(null);

  useEffect(() => {
    // Initiate model hydration sequence
    const load = async () => {
      const data = await hydrate({ version: schemaVersion });
      setCanvasState(data);
    };
    load();
  }, [nodeId, schemaVersion]);

  if (isProcessing) return <div className="loading-shimmer" />;

  return (
    <div className="node-canvas border border-border-muted rounded">
      {/* Render derived state here */}
      <CanvasRenderer data={canvasState} />
    </div>
  );
};
