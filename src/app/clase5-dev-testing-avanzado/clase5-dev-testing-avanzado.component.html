Ejemplo: Test Unitario Generado

@ExtendWith(MockitoExtension.class)
class ClienteServiceTest {{ '{' }}
    @Mock
    private ClienteRepository repository;

    @InjectMocks
    private ClienteService service;

    @Test
    void crearCliente_ConDatosValidos_DeberiaCrearCliente() {{ '{' }}
        // Given
        ClienteRequestDTO dto = ClienteRequestDTO.builder()
            .documento("12345678")
            .nombres("Juan Pérez")
            .build();

        when(repository.existsByDocumento(anyString())).thenReturn(false);

        // When
        ClienteResponseDTO resultado = service.crearCliente(dto);

        // Then
        assertThat(resultado).isNotNull();
        verify(repository).save(any(Cliente.class));
    {{ '}' }}

                {{ '{' }}"documento": "12345678", "nombres": "Juan Pérez"{{ '}' }}
