Token Estimator

Heurística rápida (TypeScript)

¿Para qué sirve?

Este snippet es una función de utilidad ligera para pre-estimar el tamaño del contexto antes de enviarlo a un LLM. Si necesitas precisión exacta, usa la librería oficial tiktoken (OpenAI).

token-utils.ts

export function estimateTokens(text: string): number {{ '{' }}
  if (!text) return 0;
  
  // 1. Separar por palabras, signos de puntuación y camelCase
  const rawPieces = text.match(/([a-zA-ZáéíóúÁÉÍÓÚñÑ]+|[0-9]+|[^\s\w]+|\s+)/g) || [];
  
  // 2. Multiplicar por un factor conservador (el español/código suele usar más tokens)
  const baseTokens = rawPieces.length;
  
  // Retornamos un 10% adicional por seguridad en la estimación
  return Math.ceil(baseTokens * 1.1);
{{ '}' }}

export function estimateCost(tokens: number, ratePerMillion: number = 0.15): number {{ '{' }}
  return (tokens / 1_000_000) * ratePerMillion;
{{ '}' }}
