Context Assembly Template

Estructura base para ensamblar payloads de contexto inyectado

context-assembly.template.xml

Role: SYSTEM 
<!-- ========================================== -->
<!-- STATIC CONTEXT (Instrucciones Inmutables)  -->
<!-- ========================================== -->
<system_instructions>
You are a highly capable agent assisting the user.
Your behavior must be concise, technical, and accurate.
</system_instructions>

<!-- ========================================== -->
<!-- DYNAMIC CONTEXT (Estado de la Aplicación)  -->
<!-- ========================================== -->
<application_context>
  <user_info id="u_8829" role="admin" />
  <current_route path="/dashboard/billing/invoices/552" />
  <page_data>
    {{ '{' }}"invoiceId": "552", "status": "unpaid", "amount": 1500{{ '}' }}
  </page_data>
</application_context>

<!-- ========================================== -->
<!-- RETRIEVED KNOWLEDGE (RAG)                  -->
<!-- ========================================== -->
<retrieved_knowledge query="how to cancel invoice">
  <document source="docs/billing.md" score="0.95">
    To cancel an unpaid invoice, admins can use the /api/invoices/:id/cancel endpoint.
  </document>
</retrieved_knowledge>

<!-- ========================================== -->
<!-- CONVERSATION HISTORY (Compreso)            -->
<!-- ========================================== -->
<conversation_history>
  <message role="user">Hi, I need help with an invoice.</message>
  <message role="assistant">I can help with that. Which invoice?</message>
</conversation_history>

<!-- ========================================== -->
<!-- RECENT USER INTENT                         -->
<!-- ========================================== -->
<user_input>
  Can I cancel this one?
</user_input>
🏗️

Patrón de Diseño

Este template usa XML tags explícitos para compartimentar el conocimiento. Esto evita que el LLM confunda una directiva del sistema con un dato de la base de datos o con una instrucción maliciosa del usuario (Prompt Injection).
